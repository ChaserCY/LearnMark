题解：（暴力检测，截取字符串）
```cpp
#include <bits/stdc++.h>
using namespace std;
int main()
{
  string s="kfdhtshmrw4nxg#f44hlbn33ccto#mwfn2waebry#3qd1ubwyhcyuavuajb#vyecsycuzsmwp31ipzah#catatja3kaqbcss2th";
  int count = 0;
  for(int i = 0;i < s.size();i++){//一个一个开始
    for(int len = 8;len<=16;len++){//长度限制
      if(i+len>s.size())break;
      string substr = s.substr(i, len);//一个符合要求子串
  
      bool number=false;
      bool str = false;
      for(char ch:substr){
         if(isdigit(ch))number=true;
        if(!isalnum(ch))str = true;
        if(number&&str)break;
      }
  
      if(number&&str)count++;
    }
  
  }
  
  cout<<count;
  return 0;
}
```
