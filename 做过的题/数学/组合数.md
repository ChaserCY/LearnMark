C(N,M)，C(n,m)=n!/((n-m)!\*m!)
表示：从N个数中随机选取M个数出来
技巧：不用计算每个数的阶乘，直接化简后来计算
代码：需要预置阶乘数，和逆阶数(费马小定理)，置换掉分母的除法部分

标准求组合数
```cpp
//快速幂处理此方
ll qpow(ll a, ll b) {  
ll res = 1;  
while(b) {  
if(b & 1) res = res * a % MOD;  
a = a * a % MOD;  
b >>= 1;  
}  
return res;  
}  

  
// 预处理阶乘和逆元  
void init(int n) {  
fact[0] = 1;  
for(int i = 1; i <= n; i++) fact[i] = fact[i-1] * i % MOD;//求正向阶乘数组(取过MOD的)  
invfact[n] = qpow(fact[n], MOD - 2);  //费马小定理，a^(−1)≡a^(MOD−2)
for(int i = n-1; i >= 0; i--) invfact[i] = invfact[i+1] * (i+1) % MOD;  //倒推
}  
  
// 组合数  
ll C(int n, int k) {  
if(k < 0 || k > n) return 0;  
return fact[n] * invfact[k] % MOD * invfact[n-k] % MOD;  
}
```
