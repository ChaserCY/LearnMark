### 问题描述

给定一个长度为 N 的整数数列：A1,A2,…,AN。你要重复以下操作 K 次：

每次选择数列中最小的整数（如果最小值不止一个，选择最靠前的），将其删除。并把与它相邻的整数加上被删除的数值。

输出 K 次操作后的序列。

### 输入格式

第一行包含两个整数 N 和 K。

第二行包含 N 个整数，A1,A2,A3,…,AN

### 输出格式

输出 N−KN−K 个整数，中间用一个空格隔开，代表 KK 次操作后的序列。

### 样例输入

```text
5 3
1 4 2 8 7
```

### 样例输出

```text
17 7
```

### 样例说明

数列变化如下，中括号里的数是当次操作中被选择的数：

[1][1] 4 2 8 7

5 [2][2] 8 7

[7][7] 10 7

17 7

### 评测用例规模与约定
对于 2020 的数据，1≤K<N≤100000
对于 100100 的数据，1≤K<N≤5×105，0≤Ai≤108
## 题解：(小顶堆+双向数组链表)
```cpp
#include <bits/stdc++.h>
using namespace std;
  
// 堆中存储的节点信息
struct Node {
    long long val; // 当前值，一定得是long long 类型的
    int idx;       // 原下标
  
    // priority_queue 默认是大根堆
    // 这里改成：值小的优先；值相同时，下标小的优先
    bool operator < (const Node& other) const {
        if (val != other.val) return val > other.val;
        return idx > other.idx;
    }
};
  
int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);
  
    int N, K;
    cin >> N >> K;
  
    vector<long long> a(N + 2);
  
    vector<int> L(N + 2), R(N + 2);
  
    vector<bool> alive(N + 2, true);//用来标记被删除的元素
  
    for (int i = 1;i <= N; i++) {
        cin >> a[i];
    }
  
    // 初始化双向链表(索引的映射)
    for (int i = 1; i <= N; i++) {
        L[i] = i - 1;
        R[i] = i + 1;
    }
    R[N] = 0; // 0 表示不存在，必须初始化这个R[N]
    alive[0] = false; //用来剔除哨兵位
  
    // 最小堆：维护当前所有“可能有效”的节点
    priority_queue<Node> pq;
    for (int i = 1; i <= N; i++) {
        pq.push({a[i], i});
    }
  
    // 删除 K 次
    for (int step = 1; step <= K; step++) {
        int x;
  
        // 懒删除：
        // 1. 这个点必须还活着
        // 2. 堆里的值必须等于当前位置当前真实值
        while (true) {
            Node cur = pq.top();
            pq.pop();
  
            if (!alive[cur.idx]) continue;      // 这个点已经被删了，跳过，//数组里的
            if (cur.val != a[cur.idx]) continue; // 检验堆的旧值，跳过。//堆里的
  
            x = cur.idx; // 找到当前真正应该删除的点
            break;
        }
  
        int l = L[x];   //找到对应元素左右索引
        int r = R[x];
  
        long long v = a[x];
  
        // 把被删除点的值加到左邻居
        if (l != 0) {
            a[l] += v;
            pq.push({a[l], l}); // 新值重新入堆，并挤掉原来的堆中索引对应元素
        }
  
        // 把被删除点的值加到右邻居
        if (r != 0) {
            a[r] += v;
            pq.push({a[r], r}); // 新值重新入堆
        }
  
        // 从链表中删除 x
        alive[x] = false;
  
        if (l != 0) R[l] = r; //实现删除后的，左右索引更新
        if (r != 0) L[r] = l;//索引不为0，才可以
    }
  
    // 输出剩余序
    int cur = 1;
    while (cur != 0 && !alive[cur]) cur = R[cur]; //找到剩余的元素第一个活着的索引
  
    while (cur != 0) {    //逐步向右遍历
        cout << a[cur];
        if (R[cur] != 0) cout << ' ';
        cur = R[cur];
    }
    cout << '\n';
  
    return 0;
}
```
